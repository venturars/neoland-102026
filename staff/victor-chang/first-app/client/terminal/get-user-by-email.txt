# Feature: getUserByEmail — practice the query in query/find-user-by-email.js
# Run this whole block in Git Bash, with first-app as the current folder.
# It uses a throwaway practice database (db/practice.db), rebuilt on every run,
# so your real data in db/users.db is never touched.

# 1. Start a fresh practice database with the users table from db/schema.sql
DB=db/practice.db
rm -f $DB
sqlite3 $DB < db/schema.sql

# 2. Add two users to practice with: Ana gets id 1, Bob gets id 2
sqlite3 $DB "INSERT INTO users (name, email) VALUES ('Ana', 'ana@x.com');"
sqlite3 $DB "INSERT INTO users (name, email) VALUES ('Bob', 'bob@x.com');"

# 3. Run the query: the user with that email
sqlite3 -header -column $DB "SELECT * FROM users WHERE email = 'ana@x.com';"

# 4. Try an email nobody has: prints nothing
sqlite3 -header -column $DB "SELECT * FROM users WHERE email = 'nobody@x.com';"
