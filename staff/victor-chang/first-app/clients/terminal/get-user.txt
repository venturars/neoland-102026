# Feature: getUser — logic/get-user.js
# Paste this whole block into Git Bash, with first-app as the current folder.
# It uses the real database, db/users.sqlite, and removes the practice user at the end.

node --no-warnings --input-type=module -e '
import { createUser } from "./logic/create-user.js";
import { getUser } from "./logic/get-user.js";
import { deleteUser } from "./logic/delete-user.js";

const practice = createUser("Practice", `practice-${Date.now()}@x.com`);

// 1. Get the user by id: prints it
console.log(getUser(practice.id));

// 2. An id nobody has: prints "User not found"
try {
  getUser(999999);
} catch (error) {
  console.log(error.message);
}

deleteUser(practice.id);
'
