# Feature: findUserByName — logic/find-user-by-name.js
# Paste this whole block into Git Bash, with first-app as the current folder.
# It uses the real database, db/users.sqlite, and removes the two practice users at the end.

node --no-warnings --input-type=module -e '
import { createUser } from "./logic/create-user.js";
import { findUserByName } from "./logic/find-user-by-name.js";
import { deleteUser } from "./logic/delete-user.js";

const name = `Practice ${Date.now()}`;
const first = createUser(name, `practice-${Date.now()}-1@x.com`);
const second = createUser(name, `practice-${Date.now()}-2@x.com`);

// 1. Find the name, with spaces around: prints both practice users, lowest id first
console.log(findUserByName(`  ${name}  `));

// 2. The same name in lowercase: prints []
console.log(findUserByName(name.toLowerCase()));

// 3. Only spaces: prints "Name is empty"
try {
  findUserByName("   ");
} catch (error) {
  console.log(error.message);
}

deleteUser(first.id);
deleteUser(second.id);
'
