# Feature: createUser — logic/create-user.js
# Paste this whole block into Git Bash, with first-app as the current folder.
# It uses the real database, db/users.sqlite, and removes the practice user at the end.

node --no-warnings --input-type=module -e '
import { createUser } from "./logic/create-user.js";
import { deleteUser } from "./logic/delete-user.js";

// 1. Create a user: prints it with its id and dates
const user = createUser("Practice", `practice-${Date.now()}@x.com`);
console.log(user);

// 2. The same email again: prints "Email is already used"
try {
  createUser("Other", user.email);
} catch (error) {
  console.log(error.message);
}

// 3. An empty name: prints "Name is empty"
try {
  createUser("   ", "empty@x.com");
} catch (error) {
  console.log(error.message);
}

// 4. An email without @: prints "Email is invalid"
try {
  createUser("Practice", "practice.x.com");
} catch (error) {
  console.log(error.message);
}

deleteUser(user.id);
'
