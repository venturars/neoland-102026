# Pega esto en la terminal, con el directorio actual en mi-primer-proyecto
node --input-type=module <<'EOF'
import { openDatabase, closeDatabase } from "./db/connection.js";
import { createSqliteUsers } from "./queries/users.js";
import { createUser } from "./logic/create-user.js";

let db;

try {
  db = await openDatabase();
  const users = await createSqliteUsers(db);
  const result = await createUser(users, "ada@example.com", "Ada");
  console.log(result);
} catch (error) {
  console.error(error.message);
} finally {
  if (db) {
    await closeDatabase(db);
  }
}
EOF
