# Pega esto en la terminal, con el directorio actual en mi-primer-proyecto
node --input-type=module <<'EOF'
import { openDatabase, closeDatabase } from "./db/connection.js";
import { createUsersTable } from "./queries/queries/create-users-table.js";
import { updateUser } from "./queries/queries/update-user.js";
import { findUserById } from "./queries/queries/find-user-by-id.js";

const db = await openDatabase();
await createUsersTable(db);
await updateUser(db, 1, {
  name: "Ada Lovelace",
  email: "ada@example.com",
  updated_at: "2026-10-19",
});
const result = await findUserById(db, 1);

console.log(result);
await closeDatabase(db);
EOF
