# Requires pnpm server running in another terminal.
# Run from the project directory after the separate initialization and creation snippets.

node --input-type=module <<'NODE'
import { updateUser } from './src/logic/update-user.js';

try {
  const user = await updateUser(1, { name: "Ada Lovelace" });
  console.log(user);
} catch (error) {
  console.error(error.message);
  process.exitCode = 1;
}
