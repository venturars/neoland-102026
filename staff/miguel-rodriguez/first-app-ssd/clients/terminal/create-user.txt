# Run from staff/miguel-rodriguez/first-app-ssd with Node 24.12.0 (nvm use 24.12.0, if available).
# First run the separate clients/terminal/create-users-table.txt example.
# This writes a contact to the working db/users.sqlite database.
# Repeating this example reports: Email is already used.
node --input-type=module <<'EOF'
let database;
try {
  const { db } = await import("./db/connection.js");
  database = db;
  const { createUser } = await import("./logic/create-user.js");
  const user = createUser("  Ada Lovelace  ", " ADA@EXAMPLE.COM ");
  console.log(user);
} catch (error) {
  console.error(error.message);
  process.exitCode = 1;
} finally {
  if (database) database.close();
}
EOF
