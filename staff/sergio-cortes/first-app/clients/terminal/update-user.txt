# Pega esto en la terminal, con el directorio actual en first-app
node --input-type=module <<'EOF'
import { openDatabase } from "./db/connections.js";
import { updateUser } from "./logic/update-user.js";
import {
  createUsersTable,
  selectUserById,
  updateUserById,
} from "./queries/user.js";

const database = openDatabase();
createUsersTable(database);

const stored = selectUserById(database, 1);
const updated = updateUser({ name: "Ada Lovelace", email: "ada@example.com" });
updateUserById(database, {
  id: stored.id,
  name: updated.name,
  email: updated.email,
});
const result = selectUserById(database, stored.id);

console.log(result);
EOF
