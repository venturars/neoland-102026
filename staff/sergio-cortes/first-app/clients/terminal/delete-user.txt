# Pega esto en la terminal, con el directorio actual en first-app
node --input-type=module <<'EOF'
import { openDatabase } from "./db/connections.js";
import { deleteUser } from "./logic/delete-user.js";
import {
  createUsersTable,
  deleteUserById,
  selectUsers,
} from "./queries/user.js";

const database = openDatabase();
createUsersTable(database);

const users = selectUsers(database);
const result = deleteUser({ id: 1, users });
deleteUserById(database, result.id);

console.log(result);
EOF
