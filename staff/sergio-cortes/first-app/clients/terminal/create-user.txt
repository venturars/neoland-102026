# Pega esto en la terminal, con el directorio actual en first-app
node --input-type=module <<'EOF'
import { openDatabase } from "./db/connections.js";
import { createUser } from "./logic/create-user.js";
import {
  createUsersTable,
  insertUser,
  selectUserById,
} from "./queries/user.js";

const database = openDatabase();
createUsersTable(database);

const created = createUser({ name: "Ada", email: "ada@example.com" });
const inserted = insertUser(database, created);
const result = selectUserById(database, inserted.lastInsertRowid);

console.log(result);
EOF
